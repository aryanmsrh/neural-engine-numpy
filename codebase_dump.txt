import numpy as np
from .utils import softmax


class Layer:
    def forward(self, *args): raise NotImplementedError
    def backward(self, *args): raise NotImplementedError


class Dense(Layer):
    def __init__(self, input_dim, output_dim):
        self.W = np.random.randn(output_dim, input_dim) * np.sqrt(2.0 / input_dim) # he init
        self.B = np.zeros((output_dim, 1))

        # stored parameter gradients and cache for backprop
        self.dW = None
        self.dB = None
        self.A_prev = None # previous layer activations
        self.Z = None # current layer pre-activations

    def forward(self, A_prev):
        self.A_prev = A_prev
        self.Z = np.dot(self.W, self.A_prev) + self.B
        return self.Z

    def backward(self, dZ): # takes error term backpropagated from subsequent layer
        m = dZ.shape[1] # number of samples
        
        self.dW = (1/m) * np.dot(dZ, self.A_prev.T) # derived formula for dW
        self.dB = (1/m) * np.sum(dZ, axis=1, keepdims=True) # sum horizontally to get (n, 1) dims

        dA_prev = np.dot(self.W.T, dZ) # pass error upward to previous hidden layer (dA^[1] in derivation)

        return dA_prev
    

class ReLU(Layer):
    def __init__(self):
        self.Z = None
    
    def forward(self, Z):
        self.Z = Z
        return np.maximum(0, Z)

    def backward(self, dA):
        return dA * (self.Z > 0) # multiply incoming dA with relu derivative (hadamard product)

    
class SoftmaxCrossEntropy(Layer):
    def __init__(self):
        self.A = None
        self.Y = None

    def forward(self, Z, Y):
        self.Y = Y
        self.A = softmax(Z, axis=0)

        m = Y.shape[1] # number of samples
        eps = 1e-15 # clip probabilities to prevent log(0)
        self.A = np.clip(self.A, eps, 1.0 - eps)
        loss = -1/m * np.sum(Y * np.log(self.A)) # categorical cross entropy loss
        return loss
    
    def backward(self):
        return self.A - self.Y # derived gradient of fused softmax + cross-entropy loss
    
# layer module complete
# note: he init keeps variance close to 1. if variance > 1, weights blow up to nans; if < 1, weights collapse to zero (vanishing gradients). we use he init for relu since non-symmetric relu cuts variance in half.

import numpy as np
import pandas as pd


def softmax(Z, axis=0):
    shift = Z - np.max(Z, axis=axis, keepdims=True)
    return np.exp(shift) / np.sum(np.exp(shift), axis=axis, keepdims=True)


def one_hot(y, classes=10):
    Y = np.zeros((classes, y.shape[0]))
    Y[y, np.arange(y.shape[0])] = 1
    return Y


def accuracy(probs, labels):
    if labels.ndim > 1:
        labels = np.argmax(labels, axis=0)
    return np.mean(np.argmax(probs, axis=0) == labels) * 100


def load_mnist(data_path="data/data.csv"):
    df = pd.read_csv(data_path)
    X = df.drop("label", axis=1).values / 255.0
    Y = df["label"].values

    m = X.shape[0]
    perm = np.random.permutation(m)
    split = int(m * 0.8)

    X_train = X[perm[:split]].T
    X_test = X[perm[split:]].T
    Y_train = one_hot(Y[perm[:split]])
    Y_test = Y[perm[split:]]

    return X_train, Y_train, X_test, Y_test

class Optimizer:
    def step(self): raise NotImplementedError

class SGD(Optimizer): # stochastic gradient descent
    def __init__(self, layers, lr):
        self.layers = layers # trainable layers (excludes loss layer)
        self.lr = lr # learning rate hyperparameter

    def step(self):
        for layer in self.layers:
            if hasattr(layer, 'W'): # update layers with learnable parameters (W, B)
                layer.W -= self.lr * layer.dW # update weights
                layer.B -= self.lr * layer.dB # update biases

import json
import numpy as np
from .utils import softmax, accuracy


class Model:
    def forward(self, X): raise NotImplementedError
    def predict(self, X): raise NotImplementedError

class Sequential(Model):
    def __init__(self, layers, optimizer, learning_rate):
        self.layers = layers[:-1]
        self.loss_layer = layers[-1]
        self.optimizer = optimizer(self.layers, learning_rate)

    def forward(self, X):
        out = X
        for layer in self.layers:
            out = layer.forward(out)
        return out

    def predict(self, X):
        return softmax(self.forward(X))

    def stream(self, X_train, Y_train, X_test, Y_test, epochs, batch_size=64):
        m = X_train.shape[1]
        total_batches = int(np.ceil(m / batch_size))

        yield {
            "type": "init",
            "total_epochs": epochs,
            "total_batches": total_batches,
            "train_samples": X_train.shape[1],
            "test_samples": X_test.shape[1]
        }

        step = 0
        test_acc, avg_loss = 0.0, 0.0

        for epoch in range(epochs):
            perm = np.random.permutation(m)
            X_shuffled = X_train[:, perm]
            Y_shuffled = Y_train[:, perm]
            epoch_loss = 0.0

            for b in range(0, m, batch_size):
                X_batch = X_shuffled[:, b:b+batch_size]
                Y_batch = Y_shuffled[:, b:b+batch_size]
                batch_num = (b // batch_size) + 1

                out = self.forward(X_batch)
                loss = self.loss_layer.forward(out, Y_batch)
                epoch_loss += loss

                grad = self.loss_layer.backward()

                for layer in reversed(self.layers):
                    grad = layer.backward(grad)

                self.optimizer.step()
                step += 1

                batch_acc = accuracy(self.loss_layer.A, Y_batch)

                yield {
                    "type": "batch_complete",
                    "step": step,
                    "batch": batch_num,
                    "loss": float(loss),
                    "batch_accuracy": float(batch_acc)
                }

            test_acc = float(accuracy(self.predict(X_test), Y_test))
            avg_loss = float(epoch_loss / total_batches)

            yield {
                "type": "epoch_complete",
                "epoch": epoch + 1,
                "test_accuracy": test_acc,
                "avg_loss": avg_loss
            }

        yield {
            "type": "finished",
            "final_accuracy": test_acc,
            "total_epochs": epochs,
            "final_loss": avg_loss
        }

    def fit(self, X_train, Y_train, X_test, Y_test, epochs, batch_size=64):
        for event in self.stream(X_train, Y_train, X_test, Y_test, epochs, batch_size):
            if event["type"] == "epoch_complete":
                print(f"epoch {event['epoch']}/{epochs} | loss: {event['avg_loss']:.4f} | test acc: {event['test_accuracy']:.2f}%")

    def save_weights(self, path="weights/model_weights.npz"):
        weights = {}
        for i, l in enumerate(self.layers):
            if hasattr(l, "W"):
                weights[f"W_{i}"] = l.W
                weights[f"B_{i}"] = l.B
        np.savez_compressed(path, **weights)

    def load_weights(self, path="weights/model_weights.npz"):
        data = np.load(path)
        for i, l in enumerate(self.layers):
            if hasattr(l, "W"):
                l.W = data[f"W_{i}"]
                l.B = data[f"B_{i}"]
        print(f"loaded weights from {path}")

    def save_stats(self, accuracy, loss, epochs, path="weights/model_stats.json"):
        dense = [l for l in self.layers if hasattr(l, "W")]
        params = sum(l.W.size + l.B.size for l in dense)
        dims = [str(l.W.shape[1]) for l in dense] + [str(dense[-1].W.shape[0])]
        stats = {
            "test_accuracy": accuracy,
            "loss": loss,
            "epochs_trained": epochs,
            "parameters": params,
            "topology": " -> ".join(dims),
            "status": "online"
        }
        with open(path, "w") as f:
            json.dump(stats, f, indent=2)

import io
import base64
import numpy as np
from PIL import Image

def process_canvas_image(img):
    gray = img.convert("L")
    bbox = gray.getbbox() # bounding box of drawn strokes
    if not bbox:
        return np.zeros((784, 1)), ""

    # crop and fit into 20x20 box preserving aspect ratio
    cropped = gray.crop(bbox)
    w, h = cropped.size
    ratio = 20.0 / max(w, h)
    new_w, new_h = max(1, int(round(w * ratio))), max(1, int(round(h * ratio)))
    resized = cropped.resize((new_w, new_h), Image.Resampling.BILINEAR)

    # paste onto 28x28 canvas centered
    canvas = Image.new("L", (28, 28), 0)
    canvas.paste(resized, ((28 - new_w) // 2, (28 - new_h) // 2))

    # center of mass shift to match mnist distribution
    arr = np.array(canvas, dtype=np.float64)
    total = np.sum(arr)
    if total > 0:
        y, x = np.indices((28, 28))
        shift_x = int(round(14.0 - np.sum(x * arr) / total))
        shift_y = int(round(14.0 - np.sum(y * arr) / total))
        arr = np.roll(arr, (shift_y, shift_x), axis=(0, 1))

    # 140x140 nearest-neighbor preview for the ui
    preview = Image.fromarray(arr.astype(np.uint8)).resize((140, 140), Image.Resampling.NEAREST)
    buf = io.BytesIO()
    preview.save(buf, format="PNG")
    b64 = "data:image/png;base64," + base64.b64encode(buf.getvalue()).decode("utf-8")

    return (arr / 255.0).reshape(784, 1), b64

def decode_base64_image(data_uri):
    return Image.open(io.BytesIO(base64.b64decode(data_uri.split(",")[-1])))

import sys
import json
from flask import Flask, request, jsonify, render_template, Response

sys.path.extend([".", "..", "web"]) # grab local modules

from preprocess import process_canvas_image, decode_base64_image
from inference import DigitRecognizer
from nn.utils import load_mnist, build_model

app = Flask(__name__, template_folder="templates")
recognizer = DigitRecognizer() # loads weights on startup

@app.route("/")
def index():
    with open("weights/model_stats.json") as f:
        stats = json.load(f)
    return render_template("index.html", stats=stats)

# sse streaming endpoint for live retraining
@app.route("/train")
def train_stream():
    epochs = int(request.args.get("epochs", 10))
    lr = float(request.args.get("lr", 0.1))

    def generate():
        X_train, Y_train, X_test, Y_test = load_mnist("data/data.csv")
        mlp = build_model()
        mlp.optimizer.lr = lr

        for event in mlp.stream(X_train, Y_train, X_test, Y_test, epochs):
            if event["type"] == "finished":
                mlp.save_weights("weights/model_weights.npz")
                mlp.save_stats(event["final_accuracy"], event["final_loss"], epochs)
                recognizer.load_weights("weights/model_weights.npz")
            yield f"data: {json.dumps(event)}\n\n"

    return Response(generate(), mimetype="text/event-stream")

@app.route("/predict", methods=["POST"])
def predict():
    img = decode_base64_image(request.json["image"])
    x_vector, preview_b64 = process_canvas_image(img) # prep 28x28 centered vector
    result = recognizer.predict(x_vector)
    result["preview"] = preview_b64
    return jsonify(result)

if __name__ == "__main__":
    port = int(sys.argv[sys.argv.index("-p") + 1]) if "-p" in sys.argv else None
    app.run(port=port)

  studioWrapper.style.display = isPanelOpen ? "block" : "none";
  btnToggleRetrain.classList.toggle("active", isPanelOpen);
  if (isPanelOpen) {
    renderGraphs();
    studioWrapper.scrollIntoView({ behavior: "smooth", block: "nearest" });
  } else if (trainEventSource) {
    stopLiveTraining();
  }

function renderGraphs() {
  const W = graphCanvas.width, H = graphCanvas.height;
  gctx.fillStyle = "#030303";
  gctx.fillRect(0, 0, W, H);
  const N = lossHistory.length;
  if (N < 2) return;

  const maxLoss = Math.max(...lossHistory, 1.0);
  gctx.strokeStyle = "#f59e0b";
  gctx.lineWidth = 1.5;
  gctx.beginPath();
  for (let i = 0; i < N; i++) {
    const x = (i / (N - 1)) * W;
    const y = (1 - lossHistory[i] / maxLoss) * (H - 10) + 5;
    i === 0 ? gctx.moveTo(x, y) : gctx.lineTo(x, y);
  }
  gctx.stroke();

  gctx.strokeStyle = "#22c55e";
  gctx.beginPath();
  for (let i = 0; i < N; i++) {
    const x = (i / (N - 1)) * W;
    const y = (1 - accHistory[i] / 100) * (H - 10) + 5;
    i === 0 ? gctx.moveTo(x, y) : gctx.lineTo(x, y);
  }
  gctx.stroke();
}

function startLiveTraining() {
  const epochs = parseInt(inpEpochs.value) || 10;
  const lr = parseFloat(inpLR.value) || 0.10;

  btnStartTrain.disabled = true;
  btnStopTrain.disabled = false;
  studioStatus.className = "phase-badge phase-forward";
  studioStatus.textContent = "training live...";

  lossHistory = [];
  accHistory = [];
  renderGraphs();

  let stepCounter = 0;
  let totalEpochs = epochs;
  let totalBatches = 0;
  let totalSteps = 0;

  trainEventSource = new EventSource(`/train?epochs=${epochs}&lr=${lr}`);

  trainEventSource.onmessage = (e) => {
    const data = JSON.parse(e.data);

    if (data.type === "init") {
      totalEpochs = data.total_epochs;
      totalBatches = data.total_batches;
      totalSteps = totalEpochs * totalBatches;
      telEpoch.textContent = `Epoch: 1/${totalEpochs}`;
      telBatch.textContent = `Batch: 1/${totalBatches}`;
    } else if (data.type === "batch_complete") {
      stepCounter = data.step;
      lossHistory.push(data.loss);
      accHistory.push(data.batch_accuracy);

      const pct = Math.min(100, (stepCounter / totalSteps) * 100);
      trainProgFill.style.width = pct + "%";
      telBatch.textContent = `Batch: ${data.batch}/${totalBatches}`;
      telLoss.textContent = `Loss: ${data.loss.toFixed(4)}`;
      telBatchAcc.textContent = `Batch Acc: ${data.batch_accuracy.toFixed(1)}%`;
      if (graphStatsReadout) graphStatsReadout.textContent = `Loss: ${data.loss.toFixed(4)} | Acc: ${data.batch_accuracy.toFixed(1)}%`;

      renderGraphs();
    } else if (data.type === "epoch_complete") {
      telTestAcc.textContent = `Test Acc: ${data.test_accuracy.toFixed(2)}%`;
      if (data.epoch < totalEpochs) telEpoch.textContent = `Epoch: ${data.epoch + 1}/${totalEpochs}`;
      renderGraphs();
    } else if (data.type === "finished") {
      stopLiveTraining();
      studioStatus.className = "phase-badge phase-forward";
      studioStatus.textContent = "completed";
      trainProgFill.style.width = "100%";

      statAccuracy.textContent = data.final_accuracy.toFixed(2) + "%";
      statLoss.textContent = data.final_loss.toFixed(4);
      statEpochs.textContent = totalEpochs;
    }
  };

  trainEventSource.onerror = () => stopLiveTraining();
}

function stopLiveTraining() {
  if (trainEventSource) {
    trainEventSource.close();
    trainEventSource = null;
  }
  btnStartTrain.disabled = false;
  btnStopTrain.disabled = true;
  if (studioStatus.textContent === "training live...") {
    studioStatus.textContent = "stopped";
    studioStatus.className = "phase-badge phase-idle";
  }
}

const drawCanvas = document.getElementById("drawCanvas");

let isDrawing = false;
let drawTimeout = null;

function clearCanvas() {
  dctx.fillStyle = "#000000";
  dctx.fillRect(0, 0, drawCanvas.width, drawCanvas.height);
}
clearCanvas();

dctx.strokeStyle = "#FFFFFF";
dctx.lineWidth = 19;
dctx.lineCap = "round";
dctx.lineJoin = "round";

function getCoords(e) {
  const rect = drawCanvas.getBoundingClientRect();
  const scaleX = drawCanvas.width / rect.width;
  const scaleY = drawCanvas.height / rect.height;
  const clientX = e.touches ? e.touches[0].clientX : e.clientX;
  const clientY = e.touches ? e.touches[0].clientY : e.clientY;
  return { x: (clientX - rect.left) * scaleX, y: (clientY - rect.top) * scaleY };
}

function startDraw(e) {
  e.preventDefault();
  isDrawing = true;
  const { x, y } = getCoords(e);
  dctx.beginPath();
  dctx.moveTo(x, y);
}

function doDraw(e) {
  if (!isDrawing) return;
  e.preventDefault();
  const { x, y } = getCoords(e);
  dctx.lineTo(x, y);
  dctx.stroke();

  if (chkLiveDraw.checked) {
    clearTimeout(drawTimeout);
    drawTimeout = setTimeout(predictDrawnCanvas, 100);
  }
}

function stopDraw() {
  if (!isDrawing) return;
  isDrawing = false;
  if (chkLiveDraw.checked) predictDrawnCanvas();
}

drawCanvas.addEventListener("mousedown", startDraw);
drawCanvas.addEventListener("mousemove", doDraw);
window.addEventListener("mouseup", stopDraw);
drawCanvas.addEventListener("touchstart", startDraw, { passive: false });
drawCanvas.addEventListener("touchmove", doDraw, { passive: false });
window.addEventListener("touchend", stopDraw);

  dctx.textBaseline = "middle";
  dctx.fillText(digit, drawCanvas.width / 2, drawCanvas.height / 2 + 10);

}

import sys
import time
import numpy as np

sys.path.extend([".", ".."])

from nn.utils import build_model


class DigitRecognizer:
    def __init__(self, weights_path="weights/model_weights.npz"):
        self.model = build_model()
        self.load_weights(weights_path)

    def load_weights(self, path):
        self.model.load_weights(path)

    def predict(self, x_vector):
        if x_vector.ndim == 1:
            x_vector = x_vector.reshape(-1, 1)

        t0 = time.perf_counter()
        probs = self.model.predict(x_vector)
        ms = (time.perf_counter() - t0) * 1000.0

        pred = int(np.argmax(probs, axis=0)[0])
        return {
            "prediction": pred,
            "confidence": float(probs[pred, 0]),
            "probabilities": [float(p) for p in probs[:, 0]],
            "inference_time_ms": round(ms, 3)
        }
